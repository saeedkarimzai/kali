    {keys:['what is javascript'],answer:'JavaScript is a programming language used to add interactive behavior to websites and web applications.'}
  ];
  for(const item of builtin)if(item.keys.some(k=>n===searchNorm(k)||n.includes(searchNorm(k))))return item.answer;
  const saved=[...kaliSearchSaved,...kaliSearchCache].reverse();
  const hit=saved.find(x=>searchNorm(x.question)===n || (n.length>5&&searchNorm(x.question).includes(n)));
  return hit?(hit.answer||hit.text||'Saved result found.') : null;
}

async function doSearch(){
  const inputEl=document.getElementById('searchquery'),q=inputEl.value.trim();if(!q)return;
  lastSearchQuestion=q;document.getElementById('searchsavebox').style.display='none';
  kaliSearchHistory.unshift({q,time:new Date().toLocaleString()});kaliSearchHistory=kaliSearchHistory.slice(0,100);localStorage.setItem(KALI_SEARCH_HISTORY_KEY,JSON.stringify(kaliSearchHistory));
  const local=localSearchAnswer(q);
  if(local){showSearchResult('Found in your offline knowledge',local,'');searchStatus('● OFFLINE RESULT · Answered from built-in or saved knowledge.');return}
  if(!navigator.onLine){showSearchResult('No offline answer found','I do not have a saved answer for this question yet. Connect to the internet and search again, or add an answer below to teach Kali Search.','');searchStatus('● OFFLINE · No matching saved answer.');document.getElementById('savequestion').textContent=q;document.getElementById('searchsavebox').style.display='block';return}
  searchStatus('Searching local knowledge, then DuckDuckGo...');
  try{
    const url='https://api.duckduckgo.com/?q='+encodeURIComponent(q)+'&format=json&no_html=1&skip_disambig=1';
    const response=await fetch(url,{mode:'cors',headers:{'Accept':'application/json'}});
    if(!response.ok)throw Error('Search service returned '+response.status);
    const data=await response.json();
    let answer=data.Answer||data.AbstractText||data.Definition||'';
    let source=data.AbstractSource||data.DefinitionSource||'DuckDuckGo';
    let link=data.AbstractURL||data.DefinitionURL||'';
    if(!answer&&Array.isArray(data.RelatedTopics)){
      const topic=data.RelatedTopics.find(x=>x.Text&&x.FirstURL)||data.RelatedTopics.flatMap(x=>x.Topics||[]).find(x=>x.Text&&x.FirstURL);
      if(topic){answer=topic.Text;link=topic.FirstURL;source='DuckDuckGo related result'}
    }
    if(answer){cacheSearchResult(q,answer,link,source);showSearchResult(source,answer,link);searchStatus('● ONLINE ANSWER · Saved to your local search cache for next time.')}
    else{showSearchResult('No instant answer returned','DuckDuckGo did not provide a short instant answer for this query. Open full search results to browse the web. You can also save your own answer below.','');searchStatus('● ONLINE · No instant answer. Full web search is available.');document.getElementById('savequestion').textContent=q;document.getElementById('searchsavebox').style.display='block'}
  }catch(e){showSearchResult('Could not retrieve an instant answer','The network is available or may be restricted, but DuckDuckGo’s instant-answer API could not be reached from this page. Open full DuckDuckGo results instead, or save an answer manually.','');searchStatus('● ONLINE SEARCH API BLOCKED OR UNAVAILABLE · Use full DuckDuckGo search.');document.getElementById('savequestion').textContent=q;document.getElementById('searchsavebox').style.display='block'}
}

async function importSearchBrain(event){
  const file=event.target.files&&event.target.files[0];if(!file)return;
  try{
    const data=JSON.parse(await file.text());
    if(!data||data.app!=='Kali Search'||!Array.isArray(data.saved)||!Array.isArray(data.cache))throw Error('That file is not a Kali Search backup.');
    const merge=(current,incoming,key)=>{const map=new Map();[...current,...incoming].forEach(x=>{if(x&&typeof x==='object'&&x[key])map.set(searchNorm(x[key]),x)});return [...map.values()]};
    kaliSearchSaved=merge(kaliSearchSaved,data.saved,'question').slice(0,500);
    kaliSearchCache=merge(kaliSearchCache,data.cache,'question').slice(0,150);
    kaliSearchHistory=Array.isArray(data.history)?[...data.history,...kaliSearchHistory].slice(0,200):kaliSearchHistory;
    localStorage.setItem(KALI_SEARCH_SAVED_KEY,JSON.stringify(kaliSearchSaved));
    localStorage.setItem(KALI_SEARCH_CACHE_KEY,JSON.stringify(kaliSearchCache));
    localStorage.setItem(KALI_SEARCH_HISTORY_KEY,JSON.stringify(kaliSearchHistory));
    showSavedSearches();searchStatus('IMPORT COMPLETE · Your USB backup is now loaded into this browser.');
  }catch(err){searchStatus('IMPORT FAILED · '+err.message)}
  event.target.value='';
}
